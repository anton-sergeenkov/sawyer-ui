# Storybook

- Category: Basic | Form | Layout | Menu | Surfaces
- Компоненты лежат в `src/ui/<категория>/<Компонент>`: `src/ui/form/UiButton` и т.д.
- Storybook для Next.js (`@storybook/nextjs-vite`): в историях работают `next/font`, `next/link`, `next/navigation` и импорт картинок, как в Next.js
- Глобальные стили (вместе со стилями Mantine) подключены в `.storybook/preview.tsx`, в историях их импортировать не нужно
- Все истории обёрнуты в `UiProvider` (провайдер Mantine) декоратором в `.storybook/preview.tsx`
- Текущий путь для `usePathname()` задаётся параметром истории: `parameters: { nextjs: { navigation: { pathname: "/contacts" } } }`

> story.stories.tsx

```tsx
import type { Meta, StoryObj } from "@storybook/nextjs-vite"

import { robotoRegular } from "../../../fonts"

import UiComponent, { options } from "./UiComponent"

const meta: Meta<typeof UiComponent> = {
  title: "Category/UiComponent",
  component: UiComponent,
  tags: ["autodocs"],
  argTypes: {
    theme: {
      options: options.theme,
      control: { type: "radio" },
    },
    isActive: {
      control: { type: "boolean" },
    },
  },
  args: {
    children: "Content",
  },
  decorators: [
    (Story) => (
      <div
        className={robotoRegular.className}
        style={{ backgroundColor: "#fff", padding: "10px" }}
      >
        <Story />
      </div>
    ),
  ],
} satisfies Meta<typeof UiComponent>

export default meta

type Story = StoryObj<typeof meta>

export const Basic: Story = {
  args: {},
}
```
